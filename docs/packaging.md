# Lucid Packaging Guide

## Overview

This document covers packaging Lucid for various distribution formats:
- **Arch Linux**: AUR (PKGBUILD)
- **Flatpak**: For universal Linux distribution
- **AppImage**: Portable single-file format

**Repository**: [github.com/ggml-org/lucid](https://github.com/ggml-org/lucid)

## Arch Linux (AUR)

### PKGBUILD

**File**: `packaging/arch/PKGBUILD`

```bash
# Maintainer: Lucid Team <dev@lucid-project.org>
# Contributor: Your Name <you@example.com>

pkgname=lucidvideo
pkgver=1.0.0
pkgrel=1
epoch=
pkgdesc="Intelligent video library with whisper.cpp and llama.cpp"
arch=('x86_64' 'aarch64')
url="https://gitlab.com/leestripp/lucid"
license=('MIT')
groups=()
depends=(
    'gtk4'
    'libadwaita'
    'glibmm'
    'gtkmm4'
    'adwaita-mm'
    'ffmpeg'
    'sqlite'
    'yt-dlp'
    'curl'
    'libcurl'
    'glu'  # For Vulkan
)
makedepends=(
    'cmake'
    'ninja'
    'meson'
    'pkgconf'
    'glm'  # GLM headers if needed
)
checkdepends=(
    'googletest'
    'valgrind'
)
optdepends=(
    'vulkan-icd: For GPU acceleration'
    'libva: For VA-API hardware decoding'
)
provides=()
conflicts=()
replaces=()
backup=()
options=(!lto)  # Disable LTO for faster builds
install=
changelog=
source=("${pkgname}-${pkgver}.tar.gz")
sha256sums=('SKIP')

prepare() {
    cd "$srcdir/$pkgname-$pkgver"
    
    # Initialize submodules
    git submodule update --init --recursive
}

build() {
    cd "$srcdir/$pkgname-$pkgver"
    
    # Create build directory
    mkdir -p build
    cd build
    
    # Configure with Vulkan for GPU acceleration
    cmake .. \
        -DCMAKE_BUILD_TYPE=Release \
        -DCMAKE_INSTALL_PREFIX=/usr \
        -DCMAKE_CXX_FLAGS="-march=x86-64 -mtune=generic" \
        -DWHISPER_VULKAN=ON \
        -DLLAMA_VULKAN=ON \
        -DENABLE_TESTS=ON \
        -GNinja
    
    # Build
    ninja
}

check() {
    cd "$srcdir/$pkgname-$pkgver/build"
    
    # Run tests
    ctest --output-on-failure || true
}

package() {
    cd "$srcdir/$pkgname-$pkgver/build"
    
    # Install executable
    install -Dm755 lucidvideo "${pkgdir}/usr/bin/lucid"
    
    # Install desktop file
    install -Dm644 "$srcdir/$pkgname-$pkgver/data/org.lucid.Lucid.desktop" \
        "${pkgdir}/usr/share/applications/org.lucid.Lucid.desktop"
    
    # Install app icon
    install -Dm644 "$srcdir/$pkgname-$pkgver/data/icons/hicolor/scalable/apps/lucidvideo.svg" \
        "${pkgdir}/usr/share/icons/hicolor/scalable/apps/org.lucid.Lucid.svg"
    
    # Install metainfo (AppStream)
    install -Dm644 "$srcdir/$pkgname-$pkgver/data/org.lucid.Lucid.metainfo.xml" \
        "${pkgdir}/usr/share/metainfo/org.lucid.Lucid.metainfo.xml"
    
    # Install GSettings schema
    install -Dm644 "$srcdir/$pkgname-$pkgver/data/org.lucid.Lucid.gschema.xml" \
        "${pkgdir}/usr/share/glib-2.0/schemas/org.lucid.Lucid.gschema.xml"
    
    # Compile GSettings schema
    glib-compile-schemas "${pkgdir}/usr/share/glib-2.0/schemas/"
    
    # Install man page (if available)
    if [ -f "$srcdir/$pkgname-$pkgver/docs/lucidvideo.1" ]; then
        install -Dm644 "$srcdir/$pkgname-$pkgver/docs/lucidvideo.1" \
            "${pkgdir}/usr/share/man/man1/lucid.1"
    fi
}
```

## Flatpak

### flatpak.yml Manifest

**File**: `packaging/flatpak/flatpak.yml`

```yaml
app-id: org.lucid.Lucid
runtime: org.gnome.Platform
runtime-version: '47'
sdk: org.gnome.Sdk
command: lucidvideo

finish-args:
  # X11 + Wayland display
  - --share=ipc
  - --socket=fallback-x11
  - --socket=wayland
  
  # GPU acceleration for Vulkan
  - --device=dri
  
  # Filesystem access for video library
  - --filesystem=home
  - --filesystem=xdg-videos
  
  # Network for YouTube downloads
  - --share=network
  
  # D-Bus for notifications
  - --talk-name=org.freedesktop.Notifications
  
  # Single-instance (optional)
  - --own-name=org.lucid.Lucid.*

modules:
  - name: lucidvideo
    buildsystem: cmake-ninja
    builddir: true
    
    config-opts:
      - -DCMAKE_BUILD_TYPE=Release
      - -DCMAKE_INSTALL_PREFIX=/app
      - -DWHISPER_VULKAN=ON
      - -DLLAMA_VULKAN=ON
      - -DENABLE_TESTS=OFF
    
    sources:
      - type: dir
        path: ../..
    
    post-install:
      # Install schema
      - install -Dm644 data/org.lucid.Lucid.gschema.xml /app/share/glib-2.0/schemas/org.lucid.Lucid.gschema.xml
      - glib-compile-schemas /app/share/glib-2.0/schemas/
      
      # Install desktop file
      - install -Dm644 data/org.lucid.Lucid.desktop /app/share/applications/org.lucid.Lucid.desktop
      
      # Install icon
      - install -Dm644 data/icons/hicolor/scalable/apps/lucidvideo.svg /app/share/icons/hicolor/scalable/apps/org.lucid.Lucid.svg
      
      # Install metainfo
      - install -Dm644 data/org.lucid.Lucid.metainfo.xml /app/share/metainfo/org.lucid.Lucid.metainfo.xml
```

### Build Flatpak

```bash
# Install flatpak-builder
sudo pacman -S flatpak flatpak-builder

# Add Flathub (if not already)
flatpak remote-add --if-not-exists flathub https://flathub.org/repo/flathub.flatpakrepo

# Install GNOME runtime
flatpak install flathub org.gnome.Platform//47 org.gnome.Sdk//47

# Build the flatpak
cd packaging/flatpak
flatpak-builder --user --force-clean --colordiff build-dir flatpak.yml

# Install locally (for testing)
flatpak-builder --user --install build-dir

# Create repo for distribution
flatpak-builder --repo=lucidvideo.flatpakrepo build-dir flatpak.yml

# Install from local repo
flatpak install lucidvideo.flatpakrepo org.lucid.Lucid
```

## AppImage

### appimage.yml

**File**: `packaging/appimage/appimage.yml`

```yaml
app: Lucid
version: 1.0.0

runtime:
  id: org.freedesktop.Platform
  version: '23.08'
  runtime: runtime
  sdk: org.freedesktop.Sdk

command: lucidvideo

ingredients:
  packages:
    - name: gtk4
    - name: libadwaita
    - name: ffmpeg
    - name: sqlite
  deb:
    - lucidvideo_1.0.0_amd64.deb
  sources:
    - type: file
      path: lucid.AppImage

script:
  - cp /app/bin/lucid /apprun/entry.sh
  - chmod +x /apprun/entry.sh
```

### Build AppImage

```bash
# Install appimage-builder
sudo pip install appimage-builder

# Build
cd packaging/appimage
appimage-builder --recipe appimage.yml

# Test
./Lucid-1.0.0-x86_64.AppImage
```

## Build System Files

### Makefile for Distribution

**File**: `packaging/Makefile`

```makefile
# Distribution build targets

.PHONY: all aur flatpak appimage clean

all: aur flatpak appimage

aur:
	cd arch && makepkg -s

flatpak:
	flatpak-builder --repo=lucidvideo.flatpakrepo build-dir flatpak.yml

appimage:
	cd appimage && appimage-builder

clean:
	rm -rf build/
	rm -f *.pkg.tar.zst
	rm -f *.flatpak
	rm -f *.AppImage

dist: all
	# Create distribution archive
	git archive --prefix=lucidvideo-1.0.0/ -o lucid-1.0.0.tar.gz HEAD
```

## Version Updates

When releasing a new version:

```bash
# 1. Update version in CMakeLists.txt
sed -i 's/version 1.0.0/version 1.1.0/' CMakeLists.txt

# 2. Update PKGBUILD
vim arch/PKGBUILD  # Update pkgver and checksums

# 3. Update flatpak.yml
sed -i 's/runtime-version: .47./runtime-version: .48./' flatpak/flatpak.yml

# 4. Create git tag
git tag -a v1.1.0 -m "Version 1.1.0 release"
git push --tags

# 5. Create GitHub release
gh release create v1.1.0 --title "Version 1.1.0" --notes "Release notes here"
```
